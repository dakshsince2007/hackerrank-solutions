# Space       O(N)
# Insight     The implementation processes commands by splitting input strings and mapping them to corresponding Python list methods.
# Interview   Before: "I would use a dictionary to map command strings to function objects." After: "This approach uses conditional branching to execute list methods, resulting in O(N * M) time complexity, where M is the cost of operations like sort or insert."
# Pitfalls    (1) The remove method raises a ValueError if the specified integer is not present in the list.  (2) The pop method raises an IndexError if called on an empty list.  (3) The insert method requires two arguments, while other commands like print or sort require none, leading to potential index errors if input parsing is not handled.
# ──────────────────────────────────────────────────

if __name__ == '__main__':
        N = int(input())
        lst=[]
        a=[]
        for i in range(N):
                b=input()
                a.append(b)
        for i in range(len(a)):
                c=a[i].split()
                if c[0]=="print":
                        print(lst)
                elif c[0]=="append":
                        lst.append(int(c[1]))
                elif c[0]=="remove":
                        lst.remove(int(c[1]))
                elif c[0]=="sort":
                        lst.sort()
                elif c[0]=="pop":
                        lst.pop()
                elif c[0]=="reverse":
                        lst.reverse()
                elif c[0]=="insert":
                        lst.insert(int(c[1]),int(c[2]))
